## Experiment 4: Patient DB & Indexing

**Hack:** Since the expected output for this experiment doesn't care about specific names, **DO NOT type 50 manual documents**. Just use a tiny 4-line `for` loop to generate all 50 documents instantly!

```javascript
use hospitalDB;

// 1. Validations (Compressed)
db.createCollection("patients", { validator: { $jsonSchema: { bsonType: "object", required: ["patientId", "name", "age", "dateOfBirth"], properties: { patientId: { bsonType: "int" }, name: { bsonType: "string" }, age: { bsonType: "int", minimum: 0, maximum: 120 }, dateOfBirth: { bsonType: "date" } } } } })
db.createCollection("doctors", { validator: { $jsonSchema: { bsonType: "object", required: ["doctorId", "name", "specialization"], properties: { doctorId: { bsonType: "int" }, name: { bsonType: "string" }, specialization: { bsonType: "string" } } } } })
db.createCollection("medicines", { validator: { $jsonSchema: { bsonType: "object", required: ["medicineId", "name", "category", "price", "expiryDate"], properties: { medicineId: { bsonType: "int" }, name: { bsonType: "string" }, category: { bsonType: "string" }, price: { bsonType: "double" }, expiryDate: { bsonType: "date" } } } } })
db.createCollection("prescriptions", { validator: { $jsonSchema: { bsonType: "object", required: ["prescriptionId", "patientId", "doctorId", "medicineId", "dateIssued"], properties: { prescriptionId: { bsonType: "int" }, patientId: { bsonType: "int" }, doctorId: { bsonType: "int" }, medicineId: { bsonType: "int" }, dateIssued: { bsonType: "date" } } } } })
db.createCollection("sales", { validator: { $jsonSchema: { bsonType: "object", required: ["saleId", "medicineId", "quantity", "saleDate"], properties: { saleId: { bsonType: "int" }, medicineId: { bsonType: "int" }, quantity: { bsonType: "int" }, saleDate: { bsonType: "date" } } } } })

// 2. Insert Data (THE ULTIMATE TIME-SAVER LOOP)
for(let i=1; i<=10; i++) {
  db.patients.insertOne({ patientId: i, name: "Patient "+i, age: 30, dateOfBirth: new Date() });
  db.doctors.insertOne({ doctorId: i, name: "Dr "+i, specialization: "General" });
  db.medicines.insertOne({ medicineId: i, name: "Med "+i, category: "pain", price: 5.0, expiryDate: new Date("2030-01-01") });
  db.prescriptions.insertOne({ prescriptionId: i, patientId: i, doctorId: i, medicineId: i, dateIssued: new Date() });
  db.sales.insertOne({ saleId: i, medicineId: i, quantity: 2, saleDate: new Date() });
}

// 3. Creating Indexes
db.patients.createIndex({ patientId: 1 })                                                  // Single
db.prescriptions.createIndex({ patientId: 1, dateIssued: -1 })                             // Compound
db.medicines.createIndex({ name: "text", category: "text" })                               // Text
db.medicines.createIndex({ expiryDate: 1 }, { partialFilterExpression: { price: { $gt: 0 } } }) // Partial
db.doctors.createIndex({ doctorId: 1 }, { unique: true })                                  // Unique

// 4. Queries using indexes
db.patients.find({ patientId: 5 })
db.prescriptions.find({ patientId: 3 }).sort({ dateIssued: -1 })
db.medicines.find({ $text: {$search: "pain" } })
db.medicines.getIndexes()

// 5. Performance Comparison
db.patients.dropIndexes()
db.patients.find({ patientId: 7 }).explain("executionStats")  // Without Index
db.patients.createIndex({ patientId: 1 })
db.patients.find({ patientId: 7 }).explain("executionStats")  // With Index

```
